39private fun imageProxyToGrayBitmap(image: ImageProxy): Bitmap {
40 val yPlane = image.planes[0]
41 val yBuffer = yPlane.buffer.duplicate()
42
43 val width = image.width
44 val height = image.height
45 val rowStride = yPlane.rowStride
46 val pixelStride = yPlane.pixelStride
47
48 // Create grayscale bitmap
49 val bitmap = createBitmap(width, height)
50 val pixels = IntArray(width * height)
51
52 var offset = 0
53 for (row in 0 until height) {
54 for (col in 0 until width) {
55 val yValue = yBuffer.get(row * rowStride + col * pixelStride).toInt() and 0xFF
56 // Create grayscale ARGB pixel (R=G=B for true grayscale)
57 pixels[offset++] = (0xFF shl 24) or (yValue shl 16) or (yValue shl 8) or yValue
58 }
59 }
60
61 bitmap.setPixels(pixels, 0, width, 0, 0, width, height)
62
63 // Apply rotation to match face detection
64 return rotateBitmap(bitmap, image.imageInfo.rotationDegrees)
65}

77private fun rotateBitmap(bitmap: Bitmap, degrees: Int): Bitmap {
78 if (degrees == 0) return bitmap
79
80 val matrix = Matrix()
81 matrix.postRotate(degrees.toFloat())
82 val rotated = Bitmap.createBitmap(bitmap, 0, 0, bitmap.width, bitmap.height, matrix, true)
83 bitmap.recycle() // Free original bitmap
84 return rotated
85}

131fun cropAndResizeFace(bitmap: Bitmap, face: Face, size: Int = ImageDefaults.FACE_CROP_SIZE): Bitmap {
132 val box = face.boundingBox
133 val x = box.left.coerceAtLeast(0)
134 val y = box.top.coerceAtLeast(0)
135 val width = box.width().coerceAtMost(bitmap.width - x)
136 val height = box.height().coerceAtMost(bitmap.height - y)
137 val cropped = Bitmap.createBitmap(bitmap, x, y, width, height)
138 return cropped.scale(size, size)
139}

154fun bitmapToGrayByteBuffer(bitmap: Bitmap): ByteBuffer {
155 val inputBuffer = ByteBuffer.allocateDirect(bitmap.width * bitmap.height * Float.SIZE_BYTES)
156 inputBuffer.order(ByteOrder.nativeOrder())
157
158 var minVal = 255f
159 var maxVal = 0f
160 var sumVal = 0f
161 var count = 0
162
163 for (y in 0 until bitmap.height) {
164 for (x in 0 until bitmap.width) {
165 val pixel = bitmap[x, y]
166 // For grayscale bitmap (R=G=B), extract any channel
167 // Using red channel for consistency
168 val gray = ((pixel shr 16) and 0xFF).toFloat()
169
170 // Normalize to [0, 1] range - MUST match Python preprocessing
171 val normalized = gray / ImageDefaults.NORMALIZE_DIVISOR
172 inputBuffer.putFloat(normalized)
173
174 // Track stats for debugging
175 if (normalized < minVal) minVal = normalized
176 if (normalized > maxVal) maxVal = normalized
177 sumVal += normalized
178 count++
179 }
180 }
181
182 inputBuffer.rewind()
183
184 // Log preprocessing stats (only occasionally to avoid spam)
185 if (count > 0 && Math.random() < ImageDefaults.LOG_PREPROCESS_SAMPLE_RATE) { // Log 1% of preprocessing
186 val avgVal = sumVal / count
187 Log.d(TAG, "Preprocessing stats - Min: $minVal, Max: $maxVal, Avg: $avgVal")
188 Log.d(TAG, "First 5 values: ${(0 until 5).map { inputBuffer.getFloat(it * Float.SIZE_BYTES) }}")
189 inputBuffer.rewind()
190 }
191
192 return inputBuffer
193}
