1"""Real-Time Emotion Detection Demo
2
3This script demonstrates real-time emotion detection from webcam using:
4- MediaPipe for face detection
5- TensorFlow Lite for emotion classification
6- OpenCV for video capture and display
7
8Features:
9- Detects faces in real-time using MediaPipe
10- Predicts emotions every 10 frames (optimization)
11- Displays bounding box and emotion label
12- Press ESC to exit
13
14Usage:
15 python video_capture_test.py
16
17Requirements:
18 - opencv-python
19 - mediapipe
20 - tensorflow
21 - numpy
22"""
23
24import cv2
25import mediapipe as mp
26from predictor import predict_emotion
27from model import load_model, emotion_label_to_string
28
29# Initialize MediaPipe Face Detection
30mp_face_detection = mp.solutions.face_detection
31mp_drawing = mp.solutions.drawing_utils
32
33# Load TensorFlow Lite model
34interpreter = load_model()
35input_details = interpreter.get_input_details()
36output_details = interpreter.get_output_details()
37
38# Initialize face detection with confidence threshold
39face_detection = mp_face_detection.FaceDetection(model_selection=0, min_detection_confidence=0.5)
40
41# Open webcam
42cap = cv2.VideoCapture(0)
43
44frame_count = 0
45last_emotion = None
46
47# Main video processing loop
48while True:
49 ret, frame = cap.read()
50 if not ret:
51 break
52
53 frame_count += 1
54
55 # Get frame dimensions and convert to RGB for MediaPipe
56 h, w, _ = frame.shape
57 frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
58 results = face_detection.process(frame_rgb)
59
60 # Process detected faces
61 if results.detections:
62 for detection in results.detections:
63 # Convert relative bounding box to absolute coordinates
64 bboxC = detection.location_data.relative_bounding_box
65 x, y, w_box, h_box = int(bboxC.xmin * w), int(bboxC.ymin * h), \
66 int(bboxC.width * w), int(bboxC.height * h)
67
68 # Apply boundary checks
69 x, y = max(0, x), max(0, y)
70 w_box, h_box = min(w - x, w_box), min(h - y, h_box)
71
72 # Predict emotion every 10 frames (performance optimization)
73 if frame_count % 10 == 0:
74 last_emotion = predict_emotion(
75 interpreter,
76 input_details,
77 output_details,
78 frame,
79 (x, y, w_box, h_box)
80 )
81
82 # Convert emotion label to string
83 emotion_string = emotion_label_to_string(last_emotion) if last_emotion is not None else "Unknown"
84
85 # Draw bounding box and emotion label
86 cv2.rectangle(frame, (x, y), (x+w_box, y+h_box), (0, 255, 0), 2)
87 cv2.putText(frame, f"Emotion: {emotion_string}({last_emotion})", (x, y - 10),
88 cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)
89
90 # Display the frame
91 cv2.imshow("Real-Time Emotion Detection", frame)
92
93 # Exit on ESC key
94 if cv2.waitKey(1) & 0xFF == 27:
95 break
96
97# Cleanup
98cap.release()
99cv2.destroyAllWindows()
