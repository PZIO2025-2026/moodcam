19def crop_face(frame, bbox):
20 """Crop face region from frame using bounding box.
21
22 Applies boundary checks to ensure cropping stays within frame bounds.
23
24 Args:
25 frame (np.ndarray): Input BGR image frame
26 bbox (tuple): Bounding box as (x, y, width, height)
27
28 Returns:
29 np.ndarray: Cropped face region
30
31 Example:
32 >>> face = crop_face(frame, (100, 100, 150, 150))
33 """
34 h, w, _ = frame.shape
35 x, y, w_box, h_box = bbox
36 x, y = max(0, x), max(0, y)
37 w_box, h_box = min(w - x, w_box), min(h - y, h_box)
38 face_crop = frame[y:y+h_box, x:x+w_box]
39 return face_crop

42def preprocess_face(face_img):
43 """Preprocess face image for model input.
44
45 Processing steps:
46 1. Convert BGR to grayscale
47 2. Resize to 48x48 pixels
48 3. Normalize to [0, 1] range (divide by 255)
49 4. Add batch and channel dimensions: (1, 48, 48, 1)
50
51 Args:
52 face_img (np.ndarray): BGR face image (any size)
53
54 Returns:
55 np.ndarray: Preprocessed face tensor of shape (1, 48, 48, 1)
56
57 Note:
58 Preprocessing must match the training pipeline exactly:
59 face_norm = face_resized / 255.0
60 """
61 face_gray = cv2.cvtColor(face_img, cv2.COLOR_BGR2GRAY)
62 face_resized = cv2.resize(face_gray, (48, 48))
63 face_norm = face_resized / 255.0
64 face_input = np.expand_dims(face_norm, axis=(0, -1)).astype(np.float32) # (1, 48, 48, 1)
